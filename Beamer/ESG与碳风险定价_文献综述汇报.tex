\end{itemize}

\note{综合 ESG 研究提供偏好机制与因子构造方法，碳风险研究进一步聚焦环境维度中的转型风险。}

\subsection{定价概念}
\begin{frame}[t]{收益关联、风险暴露与因子价格}
\vspace{.25cm}
\begin{tabularx}{\textwidth}{@{}P{2.15cm}P{4.0cm}Y@{}}
\toprule
概念 & 含义 & 对应的经验问题\\\midrule
碳排放特征 & 排放总量、增长或强度 & 排放不同的企业，收益是否不同？\\[.18cm]
碳 $\beta$ & 收益对指定碳因子的载荷 & 股票如何随碳相关组合共同变动？\\[.18cm]
因子价格 $\lambda$ & 暴露对应的预期收益差异 & 控制既有因素后，暴露能否解释预期收益？\\
\bottomrule
\end{tabularx}\par
\vspace{.24cm}
\lead{一个简化的定价表示}
\[
E(R_i^e)=\beta_{i,M}\lambda_M+\textcolor{accent}{\beta_{i,C}\lambda_C}
+\sum_{k\in X}\beta_{i,k}\lambda_k
\]
\textbf{因子载荷描述收益暴露，因子价格连接暴露与预期收益。}偏好效应也可能进入因子价格。
\vfill\source{据综述第 4.2--4.3 节及附录 E 归纳。$X$ 表示其他控制因子。}
\note{该式概括后文的暴露与预期收益关系。因子平均收益、基准模型下的 alpha、横截面因子价格各有含义。还可结合跨越检验、测试资产 alpha 联合检验和样本外表现判断定价能力。}
\end{frame}

% 05 理论
\section{理论机制}\subsection{均衡定价}
\begin{frame}[t]{绿色偏好与气候风险的定价机制}
\study{Pástor 等（2021）}
\vspace{.32cm}
\begin{columns}[T,totalwidth=\textwidth]
\column{.50\textwidth}
\lead{绿色偏好影响预期收益}
在模型的市场 ESG 中性化设定下：
\[
\mu_i=\beta_{i,M}\mu_M-\frac{\bar d}{a}g_i
\]
其他条件相同，绿色偏好提高绿色资产价格，降低其预期金融收益。
\bodygap
{\footnotesize $g_i$：绿色特征\quad $\bar d$：平均绿色偏好\\
$a$：共同风险厌恶参数}
\column{.46\textwidth}
\lead{模型中的 ESG 因子}
市场与 ESG 因子共同定价。在该模型及标准化下，ESG $\beta=g_i$，绿色减棕色因子的预期溢价为负。
\bodygap
\lead{实现收益与对冲价值}
绿色偏好意外增强时，绿色资产当期可能上涨。在气候坏状态下提供回报的资产，还具有对冲价值。
\end{columns}
\vspace{.30cm}
\textcolor{accent}{\textbf{较低的预期收益与偏好冲击下的当期上涨可以同时出现。}}
\vfill\source{[1]554,556,561--562，式（9）、（31）--（33）及气候风险扩展。}
\note{ESG beta 与绿色特征相等是模型及标准化的结果。基准模型中偏好定价与扩展模型中气候对冲分别解释。绿色股票的实际对冲表现由经验研究检验。}
\end{frame}

\begin{frame}[t]{碳 $\beta$ 定价与企业异质性}
\vspace{.22cm}

\subsection{综合判断}
\begin{frame}[t]{ESG 与碳风险的定价证据}
\vspace{.28cm}

绿色偏好与气候对冲需求影响预期收益，因子价格可以包含偏好效应。
\bodygap
\lead{经验：部分研究支持 ESG 收益与碳风险补偿}
排放特征、ESG 因子 $\alpha$ 与碳 $\beta$ 研究各有支持性发现，研究对象和证据性质有所不同。
\bodygap
\textbf{开放问题：}碳 $\beta$ 的独立解释力与跨样本稳定性仍有争议。
\bodygap
\textcolor{accent}{\textbf{后续重点：}完善暴露测量，区分溢价来源，检验跨企业与跨市场的稳定性。}
\vfill\source{文献 [1]--[9] 综合评价，对应综述第 5.4、6.1--6.2 节。}
\note{理论中 ESG 进入均衡定价，经验中部分收益与风险补偿证据成立于具体设定。综合 ESG 与碳风险提供不同层次的证据。进一步识别机制与适用范围。}
\end{frame}
