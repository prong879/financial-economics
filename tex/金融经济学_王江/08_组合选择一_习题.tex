\begin{samepage}
\begin{problem}
\WJEightExercise{exer:wj-eight-1}{} 假设有 $N$ 项资产，收益率为 $\tilde{r}_{i}$，$i=1,\cdots,N$，它们是独立同分布的。对于一个具有不满足性和凹性效用函数的投资者，计算他的最优投资组合。
\end{problem}
\end{samepage}
\Needspace{6\baselineskip}
\begin{solution}
设 $w>0$，$\theta_i$ 是投资权重，$\sum_i\theta_i=1$。沿用本章的净收益率记号，期末财富与目标函数为
\[
 \widetilde W(\theta)=w\left(1+\sum_{i=1}^N\theta_i\tilde r_i\right),\qquad
 F(\theta)=\mathrm E[u(\widetilde W(\theta))].
\]

以下比较的权重及其排列、平均产生的财富均须落在效用定义域内，期望效用也须存在。若允许卖空，仍须满足这些可行性条件。

对任意权重向量 $\theta$，将其分量作全部 $N!$ 种排列，记为 $\theta^{(\sigma)}$。独立同分布使联合收益分布具有交换对称性，所以各排列的期望效用均为 $F(\theta)$。这些排列的平均向量为 $\bar\theta=(1/N,\ldots,1/N)$。由 $u$ 的凹性和财富对权重的线性关系，
\[
 F(\bar\theta)
 =F\left(\frac1{N!}\sum_\sigma\theta^{(\sigma)}\right)
 \ge\frac1{N!}\sum_\sigma F(\theta^{(\sigma)})
 =F(\theta).
\]

因此\textbf{等权组合 $\theta_i^*=1/N$ 是最优组合之一}。这个结论适用于无约束权重及禁止卖空的单纯形：两类可行集都对排列和取平均封闭。严格凹效用加上非退化、独立的收益分布可进一步保证权重唯一；一般凹性下可存在其他同样最优的组合。\qedhere
\end{solution}

\begin{samepage}
\begin{problem}
\WJEightExercise{exer:wj-eight-2}{*} 假设你必须将 $w$ 投资于两只证券：一只无风险债券和一只风险证券。无风险债券的收益率确定为 $r_{F}$。风险证券的收益率为 $\tilde{r}$，它的均值为 $\bar{r}$ 而方差为 $\sigma^{2}$。假设你的目标就是最大化你的二次效用函数的期望值：
\[
    E\Bigl[\tilde{w}-\tfrac12 a\tilde{w}^{2}\Bigr].
\]
\begin{enumerate}[label=(\alph*)]
    \item 求解最优组合。
    \item 讨论最优组合如何依赖于股票和债券的期望收益率 $r_{F}$ 和 $\bar{r}$，股票收益率的波动率 $\sigma$，以及偏好系数 $a$。为这些依赖关系提供经济解释。
\end{enumerate}
\end{problem}
\end{samepage}
\Needspace{6\baselineskip}
\begin{solution}
\begin{enumerate}[label=(\alph*)]
\item 设股票权重为 $x$，记
\[
 R_F=1+r_F,\qquad \delta=\bar r-r_F,\qquad
 D=\sigma^2+\delta^2.
\]

取 $a>0$、$w>0$、$D>0$，并允许借贷与卖空。期末财富为
\[
 \tilde w=w\bigl(R_F+x(\tilde r-r_F)\bigr).
\]

其均值、二阶矩分别为
\[
 \mathrm E[\tilde w]=w(R_F+x\delta),\qquad
 \mathrm E[\tilde w^2]=w^2[(R_F+x\delta)^2+x^2\sigma^2].
\]

代入目标函数，得到
\[
 J(x)=w(R_F+x\delta)
 -\frac a2w^2[(R_F+x\delta)^2+x^2\sigma^2].
\]

一阶、二阶导数为
\[
 J'(x)=w\delta-aw^2(R_F\delta+xD),\qquad
 J''(x)=-aw^2D<0.
\]

故唯一无约束最优权重为
\[
 \boxed{x^*=\frac{\delta[1-awR_F]}{awD}
              =\left(\frac1{aw}-R_F\right)\frac\delta D.}
\]

债券权重为 $1-x^*$。若另有限制 $0\le x\le1$，则在严格凹目标下将上式截取到区间 $[0,1]$。退化情形 $D=0$ 时，股票与债券的收益均相同，任意权重给出相同期末财富。

\item 记 $B=1/(aw)-R_F$。固定其他参数，逐项求导得到
\[
\begin{aligned}
 \frac{\partial x^*}{\partial\bar r}
 &=B\frac{\sigma^2-\delta^2}{D^2},&
 \frac{\partial x^*}{\partial r_F}
 &=-\frac\delta D-B\frac{\sigma^2-\delta^2}{D^2},\\
 \frac{\partial x^*}{\partial\sigma^2}
 &=-\frac{B\delta}{D^2},&
 \frac{\partial x^*}{\partial a}
 &=-\frac{\delta}{a^2wD}.
\end{aligned}
\]

在 $\delta>0$、$B>0$ 时，股票权重为正，方差及偏好参数 $a$ 上升都降低股票需求；若另有 $\delta^2<\sigma^2$，股票期望收益率上升会增加需求，无风险利率上升则降低需求。$\delta^2>\sigma^2$ 时，股票期望收益率的效应反向。

例如 $w=1$、$a=0.1$、$r_F=0$、$\sigma=0.1$ 时，$\bar r$ 从 $0.2$ 增至 $0.3$，上式给出的 $x^*$ 从 $36$ 降至 $27$。二次效用的边际效用为 $1-a\tilde w$，在 $\tilde w=1/a$ 达到零；无约束最优解体现了这一饱和点。不满足性对应 $\tilde w<1/a$ 的财富区间；在这一范围内求解时，将它作为可行性约束。
这个比较使用无约束的二次目标，均值和方差不足以判断各状态的财富是否都低于饱和点。若要求不满足性在所有可能财富上成立，还须结合收益率的分布检查约束并重新求解。
\qedhere
\end{enumerate}
\end{solution}

\begin{samepage}
\begin{problem}
\WJEightExercise{exer:wj-eight-3}{*} 考虑与练习~\ref{exer:wj-eight-2}（在一只风险证券与无风险证券之间配置财富）一样的问题。不过现在假设效用函数具有常数绝对风险厌恶系数：
\[
    E\bigl[-e^{-a\tilde{w}}\bigr].
\]

另外，假设股票收益率 $\tilde{r}$ 是正态分布的。
\begin{enumerate}[label=(\alph*)]
    \item 求解最优投资组合。
    \item 讨论最优组合如何依赖于股票和债券的期望收益率 $r_{F}$ 和 $\bar{r}$，股票收益率的波动率 $\sigma$，以及偏好系数 $a$。为这些依赖性提供经济解释。[如果需要的话，你可以添加其它条件。]
\end{enumerate}
\end{problem}
\end{samepage}
\Needspace{6\baselineskip}
\begin{solution}
\begin{enumerate}[label=(\alph*)]
\item 取 $a>0$、$w>0$、$\sigma^2>0$，允许自由借贷与卖空。设股票权重为 $x$，则
\[
 \tilde w=w\bigl(1+r_F+x(\tilde r-r_F)\bigr).
\]

正态收益使期末财富也是正态变量，因此
\[
 \mathrm E[\tilde w]=w(1+r_F)+wx(\bar r-r_F),\qquad
 \operatorname{Var}(\tilde w)=w^2x^2\sigma^2,
\]
\[
 \mathrm E[-e^{-a\tilde w}]
 =-\exp\left(-a\mathrm E[\tilde w]
              +\frac{a^2}{2}\operatorname{Var}(\tilde w)\right).
\]

最大化期望效用等价于最大化确定性等值
\[
 \mathrm E[\tilde w]-\frac a2\operatorname{Var}(\tilde w)
 =w(1+r_F)+wx(\bar r-r_F)-\frac a2w^2x^2\sigma^2.
\]

对 $x$ 求导，得
\[
 w(\bar r-r_F)-aw^2x\sigma^2=0,
 \qquad x^*=\frac{\bar r-r_F}{aw\sigma^2}.
\]

二阶导数为 $-aw^2\sigma^2<0$，故这是唯一无约束最优权重。

\item 股票投资金额为
\[
 wx^*=\frac{\bar r-r_F}{a\sigma^2}.
\]

股票期望收益率上升提高股票投资；无风险利率或风险厌恶参数 $a$ 上升，降低正风险溢价下的股票投资。$\bar r>r_F$ 时，波动率上升降低正头寸的大小，财富上升使权重下降。$\bar r<r_F$ 时，最优头寸为负，波动率上升使空头金额的绝对值缩小。CARA 下股票投资金额与初始财富无关，权重则与财富成反比。
\qedhere
\end{enumerate}
\end{solution}

\begin{problem}
\WJEightExercise{exer:wj-eight-4}{**} 赋予投资者初始金融财富 $w$ 以及 $\lambda$ 份不可交易收入 $\tilde{y}$，其中 $\lambda\ge 0$ 且 $E[\tilde{y}]=0$。他可以交易一只无风险证券和一只风险证券。无风险证券的收益率为 0 而风险证券的收益率 $\tilde{r}$ 的均值为 $\bar{r}$。他具有不满足性且是风险厌恶的。他的绝对风险厌恶系数随着他的财富增加而减少。令 $a$ 为风险证券上的投资额。他的期望效用函数为：
\[
    \max E\bigl[u(w+a\tilde{r}+\lambda\tilde{y})\bigr].
\]
\begin{enumerate}[label=(\alph*)]
    \item 当 $\lambda=0$ 时，风险资产的最优头寸 $a_{0}^{*}$ 如何随着 $w$ 变化而变化？
    \item 假设 $\tilde{y}$ 和 $\tilde{r}$ 是独立的。当 $\lambda>0$ 时，记风险资产的最优头寸为 $a_{\lambda}^{*}$。$a_{\lambda}^{*}$ 是大于还是小于 $a_{0}^{*}$？
    \item 假设 $\tilde{y}=c(\tilde{r}-\bar{r})+\tilde{\varepsilon}$ 以及 $E[\tilde{\varepsilon}\mid\tilde{r}]=0$。因此，如果 $c\neq 0$，$\tilde{y}$ 和 $\tilde{r}$ 是相关的。当 $\lambda$ 极小时，他持有的股票头寸如何随着 $w$ 变化而变化？解释所得到的结论。
\end{enumerate}
\end{problem}
\end{samepage}
\Needspace{6\baselineskip}
\begin{solution}
% 原题仅假设 DARA；(b) 的背景风险比较需检查诱导效用，不能从 DARA 直接定号。
% 外部核对：Gollier--Pratt (1996), Risk Vulnerability and the Tempering Effect of Background Risk.
% https://www.tse-fr.eu/publications/journal-article/risk-vulnerability-and-tempering-effect-background-risk
以下假定所讨论的最优头寸为有限内点解，$u'>0$、$u''<0$，且相关期望和导数存在，可交换求导与期望。记 $\mu=\mathrm E[\tilde r]$。
\begin{enumerate}[label=(\alph*)]
\item 令 $W_0=w+a_0^*(w)\tilde r$，一阶条件与隐函数求导给出
\[
 \mathrm E[u'(W_0)\tilde r]=0,\qquad
 (a_0^*)'(w)=-\frac{\mathrm E[u''(W_0)\tilde r]}
                         {\mathrm E[u''(W_0)\tilde r^2]}.
\]

分母严格为负。若 $\mu>0$，则 $a_0^*>0$，由\ifdefined\WJHandoutKnowledge 定理~\ref{thm:wj-eight-3}\else 风险投资金额与财富的比较静态结果\fi 的 DARA 结论，$(a_0^*)'(w)>0$。若 $\mu<0$，将 $-\tilde r$ 看作正风险溢价的交易方向，得到 $a_0^*<0$、$(a_0^*)'(w)<0$：财富增加时空头的绝对金额增加。$\mu=0$ 时 $a_0^*=0$，财富效应为零。

\item 独立性使背景收入可以先取期望。定义
\[
 v_\lambda(t)=\mathrm E_{\tilde y}[u(t+\lambda\tilde y)],\qquad
 A_\lambda(t)=-\frac{v_\lambda''(t)}{v_\lambda'(t)}
 =\frac{\mathrm E[A(t+\lambda\tilde y)u'(t+\lambda\tilde y)]}
        {\mathrm E[u'(t+\lambda\tilde y)]}.
\]

组合选择变为 $\max_a\mathrm E_{\tilde r}[v_\lambda(w+a\tilde r)]$。$A_\lambda$ 是用边际效用加权的风险厌恶系数。DARA 描述 $A$ 随财富的变化；$A_\lambda$ 与 $A$ 的比较还取决于背景收入分布和边际效用权重。

\textbf{一个头寸增加的例子。}取效用定义域 $I=(0.8,1.15)$，并令
\[
 u(t)=(t-1)-\frac{(t-1)^2}{10}+\frac{(t-1)^3}{6}.
\]

记 $s=t-1$，则 $u'=1-s/5+s^2/2>0$、$u''=-1/5+s<0$，且
\[
 A'(t)=\frac{-24/25-s/5+s^2/2}{(1-s/5+s^2/2)^2}<0
 \quad(t\in I).
\]

令 $w=1$，$\tilde r$ 取 $0.1$、$-0.1$，概率分别为 $101/200$、$99/200$，故 $\mu=0.001$。取独立的 $\tilde y=\pm0.02$，各以 $1/2$ 的概率发生，$\lambda=1$。因为 $u'$ 为二次函数，
\[
 v_1'(t)=u'(t)+\frac{0.02^2}{2}.
\]

在原最优头寸 $a_0^*$ 处，新目标关于 $a$ 的导数为
\[
 \mathrm E[v_1'(w+a_0^*\tilde r)\tilde r]
 =\frac{0.02^2}{2}\mu>0.
\]

直接代入两状态概率，原一阶条件与加入背景风险后的一阶条件分别为
\[
 0.001-0.002a+0.000005a^2=0,\qquad
 0.0010002-0.002a+0.000005a^2=0.
\]
各取使所有财富落在 $I$ 内的根。

两目标均严格凹，最优头寸分别约为 $0.500627$、$0.500727$，所有相关财富均在 $I$ 内，因此 $a_1^*>a_0^*$。

\textbf{一个头寸减少的例子。}取 $u(t)=\log t$，仍使用上述收益分布和背景风险。此时 $a_0^*=\mu/0.1^2=0.1$。记 $b=0.02$，则
\[
 v_1'(t)=\frac{t}{t^2-b^2},\qquad
 A_1(t)=\frac{t^2+b^2}{t(t^2-b^2)}>\frac1t=A(t),\quad t>b.
\]

由\ifdefined\WJHandoutKnowledge 第~\ref{sec:wj-seven-5}~节的风险厌恶比较\else 风险厌恶比较\fi，$v_1=f\circ u$，其中 $f'>0$、$f''<0$。在 $a_0^*>0$ 处，令 $k_0=f'(u(w))$。因为 $w+a_0^*\tilde r$ 与 $\tilde r$ 同向变化，逐状态有
\[
 [f'(u(w+a_0^*\tilde r))-k_0]\,\tilde r<0.
\]

乘以正的 $u'(w+a_0^*\tilde r)$ 并取期望，再用原一阶条件消去 $k_0$ 项，得到新目标的导数严格为负，故 $a_1^*<a_0^*$。

这两个例子表明，\textbf{DARA 下，独立背景风险既可能增加风险头寸，也可能减少风险头寸}。本题的大小比较取决于具体效用与背景收入分布。若补充 $A_\lambda(t)\ge A(t)$ 在相关区间成立，则同样的凹变换论证给出：正风险溢价下 $0<a_\lambda^*\le a_0^*$；负风险溢价下空头绝对金额减小。$\mu=0$ 时两者均为零。

\item 令 $a_\lambda^*=a^*(w,\lambda)$、$W_\lambda=w+a_\lambda^*\tilde r+\lambda\tilde y$。对一阶条件求导，分别得到
\[
 \frac{\partial a^*}{\partial w}
 =-\frac{\mathrm E[u''(W_\lambda)\tilde r]}
          {\mathrm E[u''(W_\lambda)\tilde r^2]},\qquad
 \frac{\partial a^*}{\partial\lambda}
 =-\frac{\mathrm E[u''(W_\lambda)\tilde r\tilde y]}
          {\mathrm E[u''(W_\lambda)\tilde r^2]}.
\]

由 $\mathrm E[\tilde\varepsilon\mid\tilde r]=0$，在 $\lambda=0$ 处有
\[
 \mathrm E[u''(W_0)\tilde r\tilde y]
 =c\,\mathrm E[u''(W_0)\tilde r(\tilde r-\mu)].
\]

代入并利用 (a) 的财富导数，得到
\[
 \left.\frac{\partial a^*}{\partial\lambda}\right|_{\lambda=0}
 =-c[1+\mu(a_0^*)'(w)],
\]
\[
 a^*(w,\lambda)
 =a_0^*(w)-\lambda c[1+\mu(a_0^*)'(w)]+o(\lambda).
\]

收入的相关部分可写入财富为
\[
 W_\lambda=(w-\underbrace{\lambda c\mu}_{\text{确定财富调整}})
 +(a+\underbrace{\lambda c}_{\text{收入自带的证券敞口}})\tilde r
              +\lambda\tilde\varepsilon.
\]

其中 $\lambda c$ 是收入自带的风险证券敞口，$-\lambda c\mu$ 是相应的确定财富调整。若 $\mu>0$，$c>0$ 时会减少直接持股来对冲收入风险；$c<0$ 时则增加直接持股。

关于\textbf{财富效应}，若 $\mu>0$，(a) 给出 $(a_0^*)'(w)>0$。在最优解及上述导数对 $(w,\lambda)$ 连续的条件下，对固定 $w$ 存在足够小的 $\lambda\ge0$，使 $\partial a^*/\partial w>0$；$\mu<0$ 时仍为负。$\mu=0$ 时基准财富导数为零，相关背景风险下的财富效应由上面的期望表达式决定。导数的连续性保证非零基准财富效应在小扰动下保持符号。
\qedhere
\end{enumerate}
\end{solution}

\begin{samepage}
\begin{problem}
\WJEightExercise{exer:wj-eight-5}{**} 假设你是对冲基金中交易政府债的交易员。你有 $w$ 的现金额度。
\begin{itemize}
    \item 在 0 期，当美联储拍卖新债券时，你可以提交报价而买入政府债。在 1 期，你可以在二级市场上以价格 $\tilde{v}$ 再把它们卖掉，其中 $\tilde{v}=\bar{v}+\tilde{\varepsilon}$，$\bar{v}>0$ 且 $E[\tilde{\varepsilon}]=0$。假设无风险收益率为零。
    \item 你购买债券的报价是 $p$。你购买到的数量 $x(p)$ 是 $p$ 的递增函数且 $x(0)=0$。选择 $p$ 以最大化你的期望效用函数：
    \[
        V(p)=E[u(\tilde{w})],\qquad \tilde{w}=w+x(p)(\bar{v}+\tilde{\varepsilon}-p).
    \]

    假设你是不满足的且严格风险厌恶的。为简单起见，假设 $V(p)$ 是严格凹的。
\end{itemize}
\begin{enumerate}[label=(\alph*)]
    \item 写出关于最优报价 $p^{*}$ 的一阶条件。
    \item 证明 $0<p^{*}<\bar{v}$。
    \item 对于两个绝对风险厌恶系数分别为 $A_{1}(w)$ 和 $A_{2}(w)$ 的交易员，如果 $\forall\,w$，$A_{1}(w)\ge A_{2}(w)$，那么 $p_{1}^{*}\le p_{2}^{*}$，其中 $p_{i}^{*}$ 是交易员 $i$ 的最优报价，$i=1,2$。
    \item 假设 $u(\cdot)$ 是绝对风险厌恶递减的。证明初始财富 $w$ 的增加使得 $p^{*}$ 也增加。
\end{enumerate}
\end{problem}
\end{samepage}
\Needspace{6\baselineskip}
\begin{solution}
以下按题给目标函数在 $p\ge0$ 上求解。若现金额度是购债支出的硬上限，可行报价还须满足 $p\,x(p)\le w$。设 $x$ 可微且严格递增，$\tilde\varepsilon$ 非退化，相关期望及导数存在。记
\[
 W(p)=w+x(p)(\bar v+\tilde\varepsilon-p),\qquad
 q(p)=x'(p)(\bar v+\tilde\varepsilon-p)-x(p).
\]

$q(p)$ 为提高报价时的边际财富变化。
\begin{enumerate}[label=(\alph*)]
\item 内点最优报价满足
\[
 V'(p^*)=\mathrm E[u'(W(p^*))q(p^*)]=0.
\]

\item 因 $x(0)=0$，$V(0)=u(w)$。由 $x$ 在零点连续且严格递增，足够小的正报价对应趋于零的正持仓，单位债券的期望收益为 $\bar v-p>0$。若期望效用可在零持仓附近作一阶展开，则
\[
 V(p)-u(w)=u'(w)x(p)(\bar v-p)+o(x(p))>0.
\]

而在 $p=\bar v$ 时，严格 Jensen 不等式给出
\[
 V(\bar v)=\mathrm E[u(w+x(\bar v)\tilde\varepsilon)]<u(w).
\]

$p>\bar v$ 时，$\mathrm E[W(p)]<w$，所以
\[
 V(p)\le u(\mathrm E[W(p)])<u(w).
\]

因此连续的 $V$ 在 $[0,\bar v]$ 上取到最大值，最大值位于内部；严格凹性保证唯一，故 $0<p^*<\bar v$。若 $x'(0)>0$，第一步也可直接由 $V'(0)=u'(w)x'(0)\bar v>0$ 得到。

\item 由\ifdefined\WJHandoutKnowledge 第~\ref{sec:wj-seven-5}~节的 Pratt 定理\else Pratt 定理\fi，$A_1\ge A_2$ 等价于 $u_1=f\circ u_2$，其中 $f'>0$、$f''\le0$。在交易员2的最优报价 $p_2^*$ 处，记
\[
 W_2=W(p_2^*),\quad q_2=q(p_2^*),\quad
 k_0=f'(u_2(w+x(p_2^*)^2/x'(p_2^*))).
\]

由一阶条件可知 $x'(p_2^*)>0$：若它为零，则 $q_2=-x(p_2^*)<0$，与一阶条件矛盾。固定报价后，$W_2$ 与 $q_2$ 都是 $\tilde\varepsilon$ 的递增仿射函数，且 $q_2=0$ 对应财富 $w+x(p_2^*)^2/x'(p_2^*)$。因此
\[
 [f'(u_2(W_2))-k_0]q_2\le0.
\]

利用交易员2的一阶条件，得到
\[
\begin{aligned}
 V_1'(p_2^*)&=\mathrm E[f'(u_2(W_2))u_2'(W_2)q_2]\\
 &=\mathrm E[(f'(u_2(W_2))-k_0)u_2'(W_2)q_2]\le0.
\end{aligned}
\]

严格凹目标的导数递减，故 $p_1^*\le p_2^*$。

\item 取 $w_1>w_0$，令 $p_0=p^*(w_0)$、$x_0=x(p_0)$、$x'_0=x'(p_0)>0$，并记
\[
 W_0=w_0+x_0(\bar v+\tilde\varepsilon-p_0),\qquad
 q_0=x'_0(\bar v+\tilde\varepsilon-p_0)-x_0.
\]

旧一阶条件为 $\mathrm E[u'(W_0)q_0]=0$。令 $\Delta=w_1-w_0$，边际效用比为
\[
 k(t)=\frac{u'(t+\Delta)}{u'(t)},\qquad
 \frac{d}{dt}\log k(t)=A(t)-A(t+\Delta)>0.
\]

所以 $k$ 随财富递增。$q_0=0$ 对应阈值 $t_0=w_0+x_0^2/x'_0$，$W_0-t_0$ 与 $q_0$ 同号。在旧报价处增加财富后，目标的报价导数为
\[
\begin{aligned}
 V_p(p_0,w_1)
 &=\mathrm E[k(W_0)u'(W_0)q_0]\\
 &=\mathrm E[(k(W_0)-k(t_0))u'(W_0)q_0]>0.
\end{aligned}
\]

最后一步用了旧一阶条件、DARA 及非退化风险。由严格凹性，新最优报价在旧报价右侧，即 $p^*(w_1)>p^*(w_0)$。若 $V_{pp}<0$，也有 $dp^*/dw=-V_{pw}/V_{pp}>0$。
\qedhere
\end{enumerate}
\end{solution}
