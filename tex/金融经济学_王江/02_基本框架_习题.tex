\begin{problem}
\WJTwoProblem{prob:wj-two-1}{} $U(c)$ 和 $V(c)$ 是两个效用函数，$c\in\mathrm{R}_{+}^{N}$，且 $V(x)=f(U(x))$，其中 $f(\cdot)$ 是一正单调函数。证明这两个效用函数表示了相同的偏好。
\end{problem}
\begin{solution}[label-content={证明},blank-type=manual,blank-vsep=9\baselineskip]
\textbf{正单调变换}是指用严格递增的函数 $f$ 对原效用值作变换。也就是说，对 $f$ 定义域内任意 $u<v$，都有 $f(u)<f(v)$。因此，较大的效用值变换后仍然较大，相等的效用值变换后仍然相等。

设 $U$ 表示偏好 $\succeq$。对任意两个消费计划 $c_1,c_2$，结合上述性质与\ifdefined\WJHandoutKnowledge \WJRef{def:wj-two-4}{定义~}[（效用函数）]\else 教材定义 2.4\fi，有
\[
    V(c_1)\ge V(c_2)
    \iff U(c_1)\ge U(c_2)
    \iff c_1\succeq c_2.
\]

所以 $V$ 也表示偏好 $\succeq$。
\end{solution}

\begin{problem}
\WJTwoProblem{prob:wj-two-2}{} 在 1 期，经济有两个可能状态 $a$ 和 $b$，它们的发生概率相等：
\begin{center}
\WJEventTree{a,b}
\end{center}
考虑定义在消费计划 $c=\begin{bmatrix}c_{0} \\ c_{1a} \\ c_{1b}\end{bmatrix}$ 上的效用函数：
\[
    \begin{aligned}
        U(c) &= \log c_{0}+\tfrac12(\log c_{1a}+\log c_{1b})\\
        U(c) &= \tfrac{1}{1-\gamma}c_{0}^{1-\gamma}+\tfrac12\Bigl(\tfrac{1}{1-\gamma}c_{1a}^{1-\gamma}+\tfrac{1}{1-\gamma}c_{1b}^{1-\gamma}\Bigr)\\
        U(c) &= -e^{-ac_{0}}-\tfrac12\bigl(e^{-ac_{1a}}+e^{-ac_{1b}}\bigr)
    \end{aligned}
\]

证明它们满足：不满足性、连续性和凸性。
\end{problem}
\begin{solution}[label-content={证明},blank-type=manual,blank-vsep=9\baselineskip]
三个效用函数都具有形式 $U(c)=u(c_{0})+\tfrac12\bigl(u(c_{1a})+u(c_{1b})\bigr)$。对数效用要求消费严格为正，幂效用也取正的底数。指数效用在整个实数轴上有定义。为在同一个消费集上比较三种偏好，统一取
\[
    C=\bigl\{c\in\mathrm{R}^{3}:c_{0}>0,\;c_{1a}>0,\;c_{1b}>0\bigr\}
\]

三种期效用分别为
\[
    u(x)=\log x,\qquad
    u(x)=\frac{x^{1-\gamma}}{1-\gamma},\qquad
    u(x)=-e^{-ax},\qquad x>0.
\]

幂效用取 $\gamma\ge 0$ 且 $\gamma\ne 1$，指数效用取 $a>0$。若 $\gamma=1$，幂效用的原式无定义。若 $a=0$，指数效用为常数。在上述参数条件下，一阶导数均为正，二阶导数均非正，其中 $\gamma=0$ 时幂效用为线性函数。
\[
    (\log x)'=\frac1x,\quad (\log x)''=-\frac1{x^{2}},
\]
\[
    \Bigl(\frac{x^{1-\gamma}}{1-\gamma}\Bigr)'=x^{-\gamma},\quad
    \Bigl(\frac{x^{1-\gamma}}{1-\gamma}\Bigr)''=-\gamma x^{-\gamma-1},
\]
\[
    (-e^{-ax})'=ae^{-ax},\quad (-e^{-ax})''=-a^{2}e^{-ax}.
\]

\begin{enumerate}[label=(\alph*)]
    \item 不满足性（喜多厌少）。设 $c,c'\in C$ 且 $c>c'$，即各分量满足 $c_i\ge c_i'$，且至少有一个严格不等式。由 $u$ 严格递增，$u(c_i)\ge u(c_i')$，且至少一项严格较大。各项权重均为正，因此 $U(c)>U(c')$。
    \item 连续性。要证：若 $\{c^{(n)}\}\subset C$ 且 \(\lim_{n\to\infty}c^{(n)}=c\in C\)，则
    \[
        \lim_{n\to\infty}U\bigl(c^{(n)}\bigr)=U(c).
    \]

    按分量有 \(\lim_{n\to\infty}c_{i}^{(n)}=c_{i}>0\)。\(u\) 在 \((0,\infty)\) 上连续，因而在点 \(c_{i}\) 连续。由连续的序列定义，极限可以代入 \(u\)：
    \[
        \lim_{n\to\infty}u\bigl(c_{i}^{(n)}\bigr)=u\Bigl(\lim_{n\to\infty}c_{i}^{(n)}\Bigr)=u(c_{i}).
    \]
    $U$ 只是三项 $u$ 的和，极限穿过和式：
    \[
        \begin{aligned}
            \lim_{n\to\infty}U\bigl(c^{(n)}\bigr)
            &=\lim_{n\to\infty}u\bigl(c_{0}^{(n)}\bigr)
            +\tfrac12\lim_{n\to\infty}u\bigl(c_{1a}^{(n)}\bigr)
            +\tfrac12\lim_{n\to\infty}u\bigl(c_{1b}^{(n)}\bigr)\\
            &=u(c_{0})+\tfrac12\bigl(u(c_{1a})+u(c_{1b})\bigr)
            =U(c).
        \end{aligned}
    \]
    \item 凸性。$u$ 是凹函数，故对 $\alpha\in[0,1]$ 与 $c,c'\in C$，
    \[
        u\bigl(\alpha c_{i}+(1-\alpha)c_{i}'\bigr)\ge\alpha u(c_{i})+(1-\alpha)u(c_{i}'),
    \]
    且凸组合仍在 $C$ 中。因此
    \[
        U\bigl(\alpha c+(1-\alpha)c'\bigr)
        \ge\alpha U(c)+(1-\alpha)U(c')
        \ge\min\{U(c),U(c')\}.
    \]

    因此，混合消费计划至少和两者中效用较低的计划一样好，偏好满足凸性。
\end{enumerate}
\end{solution}

\begin{problem}
\WJTwoProblem{prob:wj-two-3}{} $U(c)=c-\tfrac12 a c^{2}$ 是一可能的效用函数，其中 $c\in\mathrm{R}_{+}$，$a$ 是非负的系数。$U(c)$ 具有不满足性吗？如果不是，那么 $a$ 取什么值和/或 $c$ 在什么范围内时 $U(c)$ 具有不满足性？

\begin{solution}[blank-type=manual,blank-vsep=9\baselineskip]
由 $U'(c)=1-ac$ 可知，$a=0$ 时 $U(c)=c$ 在整个非负消费域上严格递增。

若 $a>0$，则 $U$ 在 $[0,1/a]$ 上严格递增。由中值定理，即使取右端点 $1/a$，区间内任意两点的割线斜率仍等于某个正的导数值。超过 $1/a$ 后，$U'(c)<0$，消费越多，效用越低。例如 $a=1$ 时，$U(3)=-3/2<U(1)=1/2$。

\begin{problem}
\WJTwoProblem{prob:wj-two-4}{} 考虑一个经济，它在 1 期有三个可能状态：$a$，$b$ 和 $c$：
\begin{center}
\WJEventTree{a,b,c}
\end{center}
证券市场包括证券 1 和 2，它们具有如下的支付向量：$X_{1}=\begin{bmatrix}1 \\ 1 \\ 1\end{bmatrix}$ 以及 $X_{2}=\begin{bmatrix}1 \\ 2 \\ 3\end{bmatrix}$。它们的价格分别为 $S_{1}$ 和 $S_{2}$。
\begin{enumerate}[label=(\alph*)]
    \item 描述这个经济的支付空间。
    \item 写出这个经济的市场结构矩阵 $X$。
    \item 考虑含有 $\theta_{1}$ 单位的证券 1 和 $\theta_{2}$ 单位的证券 2 的组合。写出这个组合的支付向量。这个组合的价格是多少？
    \item 假设这个市场中总共有 $K$ 个参与者。每个参与者的禀赋是 1 单位的证券 1 和 2 单位的证券 2。这时的市场组合是什么？市场组合的支付向量是什么？市场组合的总价值是多少？
    \item 写出市场化支付的集合。
    \item 如果市场不允许卖空。市场化支付的集合是什么？
    \item 现在引入新的证券 3，它的支付向量为 $X_{3}=\begin{bmatrix}0 \\ 0 \\ 1\end{bmatrix}$。写出新的市场结构矩阵。在这个市场结构下，市场化支付集合是什么？
\end{enumerate}
\end{problem}
\begin{solution}[blank-type=manual,blank-vsep=9\baselineskip]
三个状态 $a,b,c$ 对应 $\Omega=3$。记
\[
    X_1=\begin{bmatrix}1 \\ 1 \\ 1\end{bmatrix},\quad
    X_2=\begin{bmatrix}1 \\ 2 \\ 3\end{bmatrix},\quad
    \theta=\begin{bmatrix}\theta_1 \\ \theta_2\end{bmatrix}.
\]

组合在 1 期的支付向量为 $X_1\theta_1+X_2\theta_2$，也可写成矩阵乘积 $X\theta$，其中 $X=[X_1,\,X_2]$。

\begin{enumerate}[label=(\alph*)]
    \item \textbf{支付空间}：所有可能在三个状态下实现的支付向量的集合，即 $\mathrm{R}^{\Omega}=\mathrm{R}^{3}$（每个分量对应状态 $a,b,c$ 下的支付）。

    \item \textbf{市场结构矩阵}：两列分别为两只证券的支付向量，
    \[
        X=\begin{bmatrix}1&1\\1&2\\1&3\end{bmatrix}=[X_1,\,X_2].
    \]

    \item \textbf{组合支付与价格：}持有 $(\theta_1,\theta_2)$ 时，
    \[
        X\theta=\theta_1 X_1+\theta_2 X_2=
        \begin{bmatrix}\theta_1+\theta_2 \\ \theta_1+2\theta_2 \\ \theta_1+3\theta_2\end{bmatrix}.
    \]

    0 期购买成本（组合价格）为 $\theta_1 S_1+\theta_2 S_2$。

    \item \textbf{市场组合：}$K$ 个参与者每人禀赋为 $1$ 单位证券 1、$2$ 单位证券 2，加总得市场组合
    \[
        \theta_M=\begin{bmatrix}K \\ 2K\end{bmatrix}.
    \]

    其支付向量为
    \[
        X\theta_M=K X_1+2K X_2=
        \begin{bmatrix}3K \\ 5K \\ 7K\end{bmatrix}.
    \]

    总价值为 $K S_1+2K S_2$。

    \item \textbf{市场化支付（允许卖空）：}能由证券组合复制的 1 期支付集合为
    \[
        M=\bigl\{Y\in\mathrm{R}^{3}: Y=X\theta=[X_1,\,X_2]\begin{bmatrix}\theta_1 \\ \theta_2\end{bmatrix}=\theta_1 X_1+\theta_2 X_2,\;\theta_1,\theta_2\in\mathrm{R}\bigr\},
    \]

    $M$ 由两只证券的支付向量张成，维数至多为 $2$。由于 $X_1$ 的三个分量相等，而 $X_2$ 的三个分量不同，两列不成比例，故 $M$ 的维数恰为 $2$。

    \item \textbf{禁止卖空：}仅允许 $\theta_1,\theta_2\ge 0$，为 $X_1,X_2$ 的\textbf{非负}线性组合：
    \[
        M_{+}=\bigl\{Y\in\mathrm{R}^{3}: Y=X\theta=[X_1,\,X_2]\begin{bmatrix}\theta_1 \\ \theta_2\end{bmatrix}=\theta_1 X_1+\theta_2 X_2,\;\theta_1,\theta_2\in\mathrm{R}_{+}\bigr\}.
    \]

    \item \textbf{加入证券 3：}其支付向量为 $X_3=\begin{bmatrix}0 \\ 0 \\ 1\end{bmatrix}$，仅在状态 $c$ 支付 $1$。两只证券时的市场结构是 $X=[X_1,\,X_2]$。加入证券 3 后的市场结构另记为 $X'$，以免与 $X$ 相混：
    \[
        X'=\begin{bmatrix}1&1&0\\1&2&0\\1&3&1\end{bmatrix}=[X_1,\,X_2,\,X_3].
    \]

    沿第三列展开得 $\det X'=1\cdot(2-1)=1$，故三列线性无关，构成 $\mathrm{R}^3$ 的一组基。允许卖空时，持仓系数可以取任意实数，所以任意 $Y\in\mathrm{R}^3$ 都可由这三只证券的组合复制。市场化支付集合为
    \[
        M'=\bigl\{Y\in\mathrm{R}^{3}: Y=\theta_1 X_1+\theta_2 X_2+\theta_3 X_3,\;\theta_1,\theta_2,\theta_3\in\mathrm{R}\bigr\}=\mathrm{R}^{3}.
    \]
\end{enumerate}
\end{solution}

\begin{problem}
\WJTwoProblem{prob:wj-two-5}{} 在\WJRef{prob:wj-two-4}{练习~} 中定义的只存在证券 1 和 2 的经济中。考虑一个禀赋为 $\theta_{1}$ 单位的证券 1 和 $\theta_{2}$ 单位的证券 2 的参与者。写出他的预算集。
\end{problem}
\begin{solution}[blank-type=manual,blank-vsep=9\baselineskip]
设 $\theta_i$ 为证券 $i$ 的初始持有量，$\alpha_i$ 为交易后的持有量，则净买入量为 $\alpha_i-\theta_i$。本题的禀赋全为证券。初始组合市值为 $w=\theta_1S_1+\theta_2S_2$，扣除新组合市值后的余额用于 0 期消费。要求 $c_0\ge0$，等价于 $\alpha_1S_1+\alpha_2S_2\le w$。

交易后的组合在 1 期提供支付 $c_1=\alpha_1X_1+\alpha_2X_2$。三个状态下的消费都须非负，即 $c_1\in\mathrm{R}_+^3$。市场允许卖空，所以单只证券的持有量 $\alpha_i$ 可以为负。完整的两期预算集为
\[
    B_{12}=\Bigl\{(c_0,c_1)\in\mathrm{R}_{+}\times\mathrm{R}_{+}^{3}:
    \begin{aligned}
        c_0&=w-\alpha_1S_1-\alpha_2S_2,\\
        c_1&=\alpha_1X_1+\alpha_2X_2,\qquad \alpha_1,\alpha_2\in\mathrm{R}
    \end{aligned}
    \Bigr\}.
\]

若只考察可实现的未来消费，可消去 $c_0$，只保留满足上述成本上限和逐状态非负约束的 $c_1$。这就是预算集在 1 期消费上的投影。
\end{solution}

\begin{problem}
\WJTwoProblem{prob:wj-two-6}{} 在上面的练习中引入\WJRef{prob:wj-two-4}{练习~} 中定义的证券 3，它的价格为 $S_{3}$。这时，参与者的预算集是什么（他在证券 3 上的禀赋为 0）？证明由证券 1、2、3 构成的预算集包含仅由证券 1、2 构成的预算集。
\end{problem}
\begin{solution}[blank-type=manual,blank-vsep=9\baselineskip]
证券 3 上的禀赋为 $0$，初始持仓的市值仍为 $w=\theta_1S_1+\theta_2S_2$。允许卖空时，加入证券 3 后的预算集为
\[
    B_{123}=\Bigl\{(c_0,c_1)\in\mathrm{R}_{+}\times\mathrm{R}_{+}^{3}:
    \begin{aligned}
        c_0&=w-\alpha_1S_1-\alpha_2S_2-\alpha_3S_3,\\
        c_1&=\alpha_1X_1+\alpha_2X_2+\alpha_3X_3,\qquad \alpha_1,\alpha_2,\alpha_3\in\mathrm{R}
    \end{aligned}
    \Bigr\}.
\]

\textbf{证明。}任取 $(c_0,c_1)\in B_{12}$，设实现该计划的持仓为 $(\alpha_1,\alpha_2)$。在新市场取 $\alpha_3=0$，则 $\alpha_3S_3=0$、$\alpha_3X_3=0$，所以买入成本和到期支付都与原组合相同，仍可实现同一个 $(c_0,c_1)$。因此 $B_{12}\subseteq B_{123}$。
\end{solution}

\begin{problem}
\WJTwoProblem{prob:wj-two-7}{} 考虑一个在 1 期只有一个可能状态的经济。（在这种情况下不存在不确定性。）参与者 1 的 0 期禀赋为 100 而 1 期禀赋为 1，即他的禀赋向量为 $\begin{bmatrix}100 \\ 1\end{bmatrix}$。他的偏好可以表示成如下形式：
\[
    U(c_{0},c_{1})=\log c_{0}+\rho\log c_{1}.
\]

系数 $\rho$ 为反映参与者在当前消费和未来消费之间相对偏好的参数。有一只证券，它的 0 期价格为 1、1 期支付为 $1+r_{F}$。这里，$r_{F}$ 是利率。
\begin{enumerate}[label=(\alph*)]
    \item 如果这个参与者不能在市场上进行交易，那么他的消费计划以及相应的效用 $U^{a}$ 是什么？
    \item 现在假设他可以在市场上进行交易。
    \begin{itemize}
        \item 他的预算集是什么？以当前消费为单位，他的总财富 $w$ 是多少？
        \item 写出参与者的优化问题。令 $c_{0}$ 为参与者的当前（即 0 期）最优消费，$s$ 为最优储蓄以及 $U^{b}$ 为在最优策略下得到的效用。求解他的最优消费/储蓄选择以及相应的效用。把 $U^{b}$ 表示成财富 $w$、利率 $r_{F}$ 和偏好系数 $\rho$ 的函数。
        \item 讨论参与者的最优选择如何依赖于利率 $r_{F}$ 和偏好系数 $\rho$。给出解释。
    \end{itemize}
    \item 证明 $U^{b}\ge U^{a}$。
    \item 令 $g$ 为参与者由于能够在证券市场上交易而获得的益处。它的定义为
    \[
        U^{b}(w-g)=U^{a}.
    \]

    计算 $g$。讨论 $g$ 如何依赖于 $\rho$？$g$ 如何依赖于 $r_{F}$？给出解释。
\end{enumerate}
\end{problem}
\begin{solution}[blank-type=manual,blank-vsep=9\baselineskip]
设 $\rho>0$。禀赋 $e_0=100$、$e_1=1$。无风险证券的 0 期价格为 $1$，1 期支付为 $1+r_F>0$。记 $S$ 为 0 期\textbf{储蓄}（买入该证券的金额，$S<0$ 表示借款）。

\begin{enumerate}[label=(\alph*)]
    \item \textbf{不能交易（自给自足）：}只能消费禀赋，
    \[
        c_0=100,\qquad c_1=1,\qquad
        U^{a}=\log(100)+\rho\log(1)=\log 100.
    \]

    \item \textbf{可以交易。}
    \begin{itemize}
        \item \textbf{预算集与财富 $w$。}买卖证券等价于在两期之间转移消费：0 期少消费 $S$ 用于储蓄，1 期增加 $S(1+r_F)$。故
        \[
            c_0=100-S,\qquad c_1=1+S(1+r_F),\qquad S\in\mathrm{R},\qquad c_0,c_1\ge0.
        \]

        以 0 期消费为计价单位，把 1 期禀赋折现到 0 期（用市场利率 $r_F$）：
        \[
            w=100+\frac{1}{1+r_F}.
        \]

        这是参与者以 0 期消费计量的\textbf{总财富}。

        \item \textbf{优化问题。}将两期预算式代入效用函数。对数的自变量必须为正，因此两期消费都要严格为正。
        \[
            \begin{aligned}
                \max_{S\in\mathrm{R}}\quad &\log(100-S)+\rho\log\bigl[1+S(1+r_F)\bigr]\\
                \text{s.t.}\quad &
                \left\{\begin{aligned}
                    100-S&>0,\\
                    1+S(1+r_F)&>0.
                \end{aligned}\right.
            \end{aligned}
        \]

        由于 $1+r_F>0$，两条约束合起来给出 $-\dfrac{1}{1+r_F}<S<100$。

        储蓄增加一单位，会少消费一单位当期商品，并增加 $1+r_F$ 单位未来消费。一阶条件为
        \[
            \underbrace{\frac{1}{100-S}}_{\text{0 期边际效用}}
            =(1+r_F)\underbrace{\frac{\rho}{1+S(1+r_F)}}_{\text{1 期边际效用}}.
        \]

        整理得 $\bigl[1+S(1+r_F)\bigr]=\rho(1+r_F)(100-S)$，即
        \[
            1+S(1+r_F)(1+\rho)=100\rho(1+r_F).
        \]
        
        其解记为最优储蓄
        \[
            S^{*}=\frac{100\rho(1+r_F)-1}{(1+\rho)(1+r_F)}.
        \]

        对任意可行的储蓄量 $S$，两期预算约束为 $c_0=100-S$ 和 $c_1=1+S(1+r_F)$。将 $S^*$ 代入两式，得到最优消费
        \[
            \begin{aligned}
                c_0^{*}&=100-S^{*}
                =100-\frac{100\rho(1+r_F)-1}{(1+\rho)(1+r_F)}
                =\frac{100(1+r_F)+1}{(1+\rho)(1+r_F)},\\[0.4em]
                c_1^{*}&=1+S^{*}(1+r_F)
                =1+(1+r_F)\cdot\frac{100\rho(1+r_F)-1}{(1+\rho)(1+r_F)}
                =\frac{\rho\bigl[100(1+r_F)+1\bigr]}{1+\rho}.
            \end{aligned}
        \]

        财富 \(w=100+\dfrac{1}{1+r_F}=\dfrac{100(1+r_F)+1}{1+r_F}\) 的分子正是上式的 \(100(1+r_F)+1\)，故
        \[
            c_0^{*}=\frac{w}{1+\rho},\qquad
            c_1^{*}=\frac{\rho(1+r_F)}{1+\rho}\,w.
        \]

        两式相除，得到两期最优消费的关系
        \[
            c_1^{*}=\rho(1+r_F)\,c_0^{*}.
        \]

        两期消费均严格为正，且目标函数严格凹，因此这是唯一最优解。按 0 期消费计价，两期的支出份额分别是 $c_0^*/w=1/(1+\rho)$ 和 $[c_1^*/(1+r_F)]/w=\rho/(1+\rho)$，只取决于 $\rho$。代入效用函数得到
        \[
            U^{b}=\log\Bigl(\frac{w}{1+\rho}\Bigr)+\rho\log\Bigl(\frac{\rho(1+r_F)}{1+\rho}\,w\Bigr).
        \]

        \item \textbf{比较静态。}上面已经求出最优消费量：
        \[
            c_0^{*}=\frac{100(1+r_F)+1}{(1+\rho)(1+r_F)}
            =\frac{1}{1+\rho}\Bigl(100+\frac{1}{1+r_F}\Bigr),\qquad
            c_1^{*}=\frac{\rho\bigl[100(1+r_F)+1\bigr]}{1+\rho}.
        \]
        \(r_F\) 只出现在 \(c_0^{*}\) 的 \(1/(1+r_F)\) 里，故 \(r_F\) 上升时 \(c_0^{*}\) 下降；\(\rho\) 在分母，\(\rho\) 上升时 \(c_0^{*}\) 也下降。
        
        把 $c_1^*$ 写成 $\frac{\rho}{1+\rho}[100(1+r_F)+1]$。固定 $\rho$ 时，后一个因子随 $r_F$ 增大。固定 $r_F$ 时，前一个因子随 $\rho$ 增大。因此，单独提高任一参数都会使 $c_1^*$ 上升。
        
        由 $S^*=100-c_0^*$，提高任一参数都会增加最优储蓄。利率升高提高了储蓄回报，$\rho$ 升高则提高了未来消费的权重。
    \end{itemize}

    \item \textbf{证明。}\textbf{$U^{b}\ge U^{a}$。}不能交易时只能消费禀赋，\((c_0,c_1)=(100,1)\)，效用为 \(U^{a}=\log 100\)。
    
    能交易时 \(S=0\) 仍可行，对应的消费还是 \((100,1)\)，效用仍是 \(\log 100=U^{a}\)。\(U^{b}\) 是所有可行储蓄里的最大效用，而 \(S=0\) 只是其中一个可行点，最大值不会低于这一点：
    \[
        U^{b}=\max_{-1/(1+r_F)<S<100}\,U\bigl(100-S,\,1+S(1+r_F)\bigr)
        \ge U(100,1)=U^{a}.
    \]

    由最优解的唯一性，等号当且仅当 $S^*=0$，即 $100\rho(1+r_F)=1$。

    \item \textbf{交易益处 $g$。}记 $U^b(z)$ 为拥有 0 期财富 $z$ 时，通过交易可达到的最高效用。令 $\hat w$ 满足 $U^b(\hat w)=U^a$，即允许交易时，只需财富 $\hat w$ 就能达到自给自足的效用。交易益处 $g=w-\hat w$ 衡量在保有交易机会的情况下，可以放弃多少 0 期财富而仍达到 $U^a$。

    将 $U^{a}=\log 100$ 与上式 $U^{b}$ 联立，记 $w-g=\hat w$，则
    \[
        (1+\rho)\log\hat w+\rho\log\bigl(\rho(1+r_F)\bigr)-(1+\rho)\log(1+\rho)=\log 100.
    \]

    取指数并整理：
    \[
        \hat w^{1+\rho}\,\frac{\bigl[\rho(1+r_F)\bigr]^{\rho}}{(1+\rho)^{1+\rho}}=100
        \quad\Rightarrow\quad
        \hat w=\rho^{-\frac{\rho}{1+\rho}}(1+r_F)^{-\frac{\rho}{1+\rho}}
        100^{\frac{1}{1+\rho}}(1+\rho).
    \]

    故
    \[
        g=w-\hat w
        =w-\rho^{-\frac{\rho}{1+\rho}}(1+r_F)^{-\frac{\rho}{1+\rho}}
        100^{\frac{1}{1+\rho}}(1+\rho).
    \]

    % 学生版不写入隐藏图注的标签。
    \IfRefUndefinedExpandable{fig:wj-gains-from-trade}{}{\WJRef{fig:wj-gains-from-trade}{图~} 中，先从 $U^a$ 水平移到 $U^b(z)$ 曲线，再向下读出 $\hat w$。横轴上 $\hat w$ 到 $w$ 的距离就是 $g$。}

    \textbf{比较静态。}由上面的效用等式，
    \[
        \log\hat w=\log(1+\rho)
        +\frac{\log 100-\rho\log[\rho(1+r_F)]}{1+\rho}.
    \]

    对右边求偏导，再乘以 $\hat w$，得到 $\hat w$ 的偏导。由 $g=w-\hat w$ 及 $w=100+1/(1+r_F)$，可得
    \[
        \frac{\partial g}{\partial \rho}
        =\frac{\hat w\log\!\bigl[100\rho(1+r_F)\bigr]}{(1+\rho)^2},
        \qquad
        \frac{\partial g}{\partial r_F}
        =\frac{\bigl[100\rho(1+r_F)\bigr]^{1/(1+\rho)}-1}{(1+r_F)^2}.
    \]

    当 $100\rho(1+r_F)<1$ 时，参与者最优选择借款，单独提高 $\rho$ 或 $r_F$ 会使 $g$ 下降，直到 $100\rho(1+r_F)=1$。此时最优选择恰好是不交易，$g=0$。越过这一点后，参与者选择储蓄，继续提高任一参数会使 $g$ 上升。
\end{enumerate}
\end{solution}
