\section{一般市场结构}\label{sec:wj-four-market}
\subsection{复合证券与支付矩阵}
\begin{tcolorbox}[knowledge,title=复合证券]
在多个状态下有非零支付的证券称为\textbf{复合证券}。
\end{tcolorbox}
\begin{tcolorbox}[knowledge,title=支付矩阵与行列约定,breakable=false]
设 1 期有 $\Omega$ 个状态，市场上有 $N$ 只证券（$n=1,\dots,N$），第 $n$ 只证券的支付向量为 $X_n=\begin{bmatrix}x_{1,n} \\ \vdots \\ x_{\Omega,n}\end{bmatrix}$，\textbf{支付矩阵}以状态为行、证券为列：
\[
    X=[X_{\cdot,1},\,\cdots,\,X_{\cdot,N}]=
    \begin{bmatrix}
        x_{1,1}   & \cdots & x_{1,N}   \\
        \vdots    & \ddots & \vdots    \\
        x_{\omega,1} & \cdots & x_{\omega,N} \\
        \vdots    & \ddots & \vdots    \\
        x_{\Omega,1} & \cdots & x_{\Omega,N}
    \end{bmatrix}.
\]
\end{tcolorbox}

\subsection{冗余证券与复制组合}
\begin{tcolorbox}[knowledge,title=冗余证券]
若证券 $j$ 的支付列 $x_j$ 可由其余证券的\textbf{复制组合}（replicating portfolio）线性表出，即存在固定组合 $\theta^*_{\setminus j}$ 使得
\[
    x_j=X_{\setminus j}\theta^*_{\setminus j},
\]
则称 $j$ 为\textbf{支付意义上的冗余证券}。
\end{tcolorbox}
对任意原组合 $\theta$，
\[
 X\theta=X_{\setminus j}\theta_{\setminus j}+\theta_jx_j
 =X_{\setminus j}(\theta_{\setminus j}+\theta_j\theta^*_{\setminus j}),
\]
其中 $\theta^*_{\setminus j}$ 是复制 $x_j$ 的固定配方。这说明删去冗余列不改变可达支付集合。若市场无摩擦且满足一价定律，还必须有
\[
 S_j=S_{\setminus j}^{\mathsf T}\theta^*_{\setminus j},
\]

所以替换前后的支付与成本相同。相应地将证券禀赋换算为复制组合，就得到相同的消费机会。交易成本或卖空限制会改变实际可实施的组合集合。

% 来源：教师 PPT 第 4—5 页。
\begin{tcolorbox}[example,title=课堂例题：第三只证券是否增加支付维度]
\begin{question}
晴、雨两个状态下，证券 $A,B,C$ 的支付为
\[
 X=\begin{bmatrix}10&0&20\\0&10&10\end{bmatrix}.
\]

判断证券 $C$ 是否冗余，并说明其价格与 $A,B$ 的关系。
\end{question}
\begin{solution}
$C=2A+B$，故第三列虽不同于前两列，却没有增加独立支付维度；无套利价格必须满足 $S_C=2S_A+S_B$。例如原来持有 $aA+bB+cC$，可改为 $(a+2c)A+(b+c)B$，各状态支付与成本均相同。增加独立支付方向才会扩大市场可生成的支付空间。
\end{solution}
\end{tcolorbox}

\subsection{证券市场的等价描述}
\begin{tcolorbox}[knowledge,title=证券市场的等价基底,breakable=false]
本小节删去冗余证券，将保留的证券只数仍记为 $N$。此时 $X$ 的各列线性无关，$\operatorname{rank}(X)=N\le\Omega$。任取 $N$ 个线性无关的组合向量 $\theta_1,\dots,\theta_N$，令 $H=[\theta_1,\cdots,\theta_N]$（$N\times N$ 可逆），则新支付矩阵 $X_\theta=XH$ 与原市场\textbf{等价}：原证券列空间不变，$X=X_\theta H^{-1}$，每只原证券都可由新组合复制。
\end{tcolorbox}

更一般地，原组合 $\theta$ 对应新组合 $H^{-1}\theta$，因为
\[
 X\theta=XH(H^{-1}\theta)=X_\theta(H^{-1}\theta).
\]

新证券的价格须同时变为 $S_\theta=H^{\mathsf T}S$，于是
\[
 S_\theta^{\mathsf T}H^{-1}\theta=S^{\mathsf T}\theta.
\]

这一变换同时保持支付与成本，相当于更换支付空间的一组基。秩不变也可由 $X=XHH^{-1}$ 得到
\[
 \mathrm{rank}(X)\le\mathrm{rank}(XH)\le\mathrm{rank}(X),
 \qquad\mathrm{rank}(XH)=N.
\]

\subsection{生成与完全市场}\label{sec:wj-four-complete}
对一般支付矩阵 $X\in\mathrm R^{\Omega\times N}$，记 $\mathbf{1}_\omega$ 为第 $\omega$ 分量为 1、其余为 0 的列向量（状态 $\omega$ \hyperref[def:wj-three-claim]{阿罗证券}的支付）。若 $\mathrm{rank}(X)=\Omega$ 且 $N\ge\Omega$，则对每一 $\omega$ 存在组合 $\theta_\omega$ 使 $X\theta_\omega=\mathbf{1}_\omega$。当 $X$ 为 $\Omega\times\Omega$ 可逆方阵时，解唯一：
\[
    \theta_\omega=X^{-1}\mathbf{1}_\omega.
\]
此时证券支付\textbf{生成}（span）整个状态支付空间，市场与完全的阿罗证券市场具有相同的可达支付集合。

\begin{tcolorbox}[knowledge]
\WJFourTheorem{thm:wj-four-1}证券市场完全，当且仅当 $\operatorname{rank}(X)=\Omega$，即独立支付方向的个数等于状态数。删去冗余证券后，可取 $N=\Omega$，此时支付矩阵为可逆方阵。
\end{tcolorbox}

\textbf{（完全市场 $\Rightarrow$ 独立支付数等于状态数）}每个单位状态支付都能复制，故 $X$ 的列空间包含 $\mathrm R^\Omega$ 的一组基，其维数为 $\Omega$。

\textbf{（独立支付数等于状态数 $\Rightarrow$ 完全市场）}这些独立列构成 $\mathrm R^\Omega$ 的基，每个目标支付都可由它们线性表示。有限价格下，相应组合的成本也有限，满足定义~\ref{def:wj-three-complete}（任一有限消费计划均可通过有限成本的证券组合融资）。这也是第~\ref{sec:wj-three-rank} 节的行满秩判据。

\begin{tcolorbox}[example,title=课堂例题：先复制阿罗证券，再复制任意支付]
\begin{question}
两状态、两证券的支付矩阵为
\[
 X=\begin{bmatrix}2&3\\5&1\end{bmatrix},\qquad
 X^{-1}=\frac1{13}\begin{bmatrix}-1&3\\5&-2\end{bmatrix}.
\]

求复制两只阿罗证券及支付 $y=(5,2)^{\mathsf T}$ 的组合与成本。
\end{question}
\begin{solution}
由 $X^{-1}$ 的两列，$\theta_1=[-1,5]^{\mathsf T}/13$、$\theta_2=[3,-2]^{\mathsf T}/13$ 分别复制 $[1,0]^{\mathsf T}$、$[0,1]^{\mathsf T}$。要复制支付 $y=[5,2]^{\mathsf T}$，先把它写成五份第一状态阿罗证券加两份第二状态阿罗证券，再换成可交易证券：
\[
 \theta=X^{-1}y=5\theta_1+2\theta_2
 =\frac1{13}\begin{bmatrix}1\\21\end{bmatrix},\qquad
 X\theta=\frac1{13}\begin{bmatrix}2+63\\5+21\end{bmatrix}
 =\begin{bmatrix}5\\2\end{bmatrix}.
\]

复制成本为 $(S_1+21S_2)/13$。阿罗证券不必实际挂牌交易，能用已有证券复制即可。
\end{solution}
\end{tcolorbox}

\section{套利}\label{sec:wj-four-arbitrage}
\subsection{组合成本与套利的定义}
\begin{tcolorbox}[knowledge,title=组合成本、支付与有限责任,breakable=false]
记交易证券的 0 期价格向量为 $S=\begin{bmatrix}S_1 \\ \vdots \\ S_N\end{bmatrix}$。组合 $\theta=\begin{bmatrix}\theta_1 \\ \vdots \\ \theta_N\end{bmatrix}$ 的 0 期成本为 $S^{\mathsf T}\theta$，1 期支付为 $X\theta$。从支付 $X$ 到价格 $S$ 的映射称为\textbf{资产定价关系}（资产定价模型）。若 $X\theta\ge 0$，称该组合在 1 期满足\textbf{有限责任}。
\end{tcolorbox}

\begin{tcolorbox}[knowledge,title=向量不等号与净现金流,breakable=false]
对列向量 $z$，$z\ge0$ 指各分量非负，$z>0$ 指 $z\ge0$ 且 $z\ne0$（至少一个分量为正），$z\gg0$ 才指每个分量都严格为正。套利的净现金流在各时点、各状态均非负，且至少一项严格为正。
\end{tcolorbox}

\begin{tcolorbox}[knowledge]
\WJFourDefinition{def:wj-four-1}组合 $\theta$ 构成\textbf{套利机会}，若同时满足：
\begin{enumerate}
    \item $S^{\mathsf T}\theta\le 0$（0 期净支出非正）；
    \item $X\theta\ge 0$（1 期各状态支付非负）；
    \item 上述两式中\textbf{至少有一个严格不等式}成立（0 期严格收入或某状态严格正支付）。
\end{enumerate}
等价地，扩展支付向量 $\begin{bmatrix}-S^{\mathsf T}\theta \\ X\theta\end{bmatrix}$ 非负且至少一分量严格为正。
\end{tcolorbox}

\subsection{三类套利与融资方式}
\begin{tcolorbox}[knowledge,title=三类套利,breakable=false]
套利分为三类：
\begin{itemize}
    \item \textbf{第 1 类套利：}$S^{\mathsf T}\theta<0$ 且 $X\theta=0$——当期净收入为正、未来净支付为零；
    \item \textbf{第 2 类套利：}$S^{\mathsf T}\theta=0$ 且 $X\theta>0$——\textbf{套利组合}（零成本、未来非负且非零支付）；
    \item \textbf{第 3 类套利：}$S^{\mathsf T}\theta<0$ 且 $X\theta>0$——0 期收入为正且未来非负、非零。
\end{itemize}
\end{tcolorbox}

初始成本为零时，买入恰由卖空收入融资；初始成本为负时，卖空收入支付买入成本后还有余额。这两种交易都由组合本身提供初始资金。

\begin{tcolorbox}[example]
\begin{question}
\WJFourExample{ex:wj-four-1}三只证券的价格写在树的根上，支付写在各枝：
\begin{center}
\begin{tabular}{@{}c@{\quad}c@{\quad}c@{}}
\textbf{A} & \textbf{B} & \textbf{C} \\[0.35em]
\WJEventTree[left={$1$},stem]{1,1,1} &
\WJEventTree[left={$1$},stem]{0,2,2} &
\WJEventTree[left={$2$},stem]{2,0,0}
\end{tabular}
\end{center}
即 $S^{\mathsf T}=[1,\,1,\,2]$，$X=\begin{bmatrix}1&0&2\\1&2&0\\1&2&0\end{bmatrix}$。找出第 1 类和第 2 类套利组合。
\end{question}
\begin{solution}

组合 $\theta_1=[2;\,-1;\,-1]$：买入 $2$ 单位 $A$、卖出 $1$ 单位 $B$ 和 $1$ 单位 $C$。$S^{\mathsf T}\theta_1=-1$，$X\theta_1=0$，为\textbf{第 1 类套利}。支付 $[2;2;2]$ 可由 $2$ 单位 $A$ 以成本 $2$ 复制，也可由 $1$ 单位 $B$ 加 $1$ 单位 $C$ 以成本 $3$ 复制——同支付不同价。

组合 $\theta_2=[2;\,0;\,-1]$：买入 $2$ 单位 $A$、卖出 $1$ 单位 $C$。$S^{\mathsf T}\theta_2=0$，$X\theta_2=[0;\,2;\,2]>0$，为\textbf{第 2 类套利}。
\end{solution}
\end{tcolorbox}

\section{无套利原理}\label{sec:wj-four-no-arbitrage}
\begin{tcolorbox}[knowledge]
\WJFourDefinition{def:wj-four-2}证券市场上不存在套利机会，称为满足\textbf{无套利原理}（no-arbitrage principle）。
\end{tcolorbox}
\subsection{均衡排除套利}
套利机会只依赖公开的支付 $X$ 与价格 $S$，不依赖私人信息或对状态概率的主观信念。

\begin{tcolorbox}[knowledge]
\WJFourTheorem{thm:wj-four-2}在无摩擦市场中，若至少一名参与者对每个非负、非零的消费增量均严格偏好，则\hyperref[sec:wj-three-equilibrium]{市场均衡}中不存在套利机会。
\end{tcolorbox}

% 来源：教材第 54 页、教师 PPT 第 15—20 页。以一次消费改进证明最优性矛盾。
\textbf{证明。}反设均衡价格下存在套利 $\theta$。给任一参与者原有证券头寸再加上 $\theta$，同时把初始净收入用于 0 期消费，消费可由 $c_k$ 提高为
\[
 c'_k=c_k+\begin{bmatrix}-S^{\mathsf T}\theta\\X\theta\end{bmatrix}
 \ge c_k,\qquad c'_k\ne c_k.
\]

增加的消费非负且非零。取满足定理单调性条件的参与者，便有 $U_k(c'_k)>U_k(c_k)$，与其均衡消费最优矛盾。

\subsection{价格约束与市场均衡}
市场均衡同时要求个人最优与市场出清，无套利是其中的必要条件。信息有效性考察价格对信息的反映程度，本章的无套利条件则直接比较逐状态支付与价格。无摩擦描述买卖与融资的交易条件，无套利约束给定支付下的证券价格。例如，例~\ref{ex:wj-four-1} 的市场允许自由交易，但其报价产生套利。
% PPT 第 17 页将无套利与无摩擦等同；此处按制度条件与价格约束区分。

% 来源：教师 PPT 第 18—20 页。
\begin{tcolorbox}[example,title=课堂例题：跨市场的一价定律]
设同一时刻、同一可交割黄金在两个市场分别报价 $1900$ 与 $1905$ 美元。假设可同时成交、自由卖空和交割，且没有交易与运输成本。买入低价黄金、卖出高价黄金，立即收入 $5$ 美元，净实物头寸为零，构成第 1 类套利。
\end{tcolorbox}
买入使低价市场的需求增加，卖出使高价市场的供给增加，价差因此受到收窄的压力。有交易成本时，可执行的净价差须扣除交易、运输等成本。
% PPT 第 19 页的中点报价是调整示意，均衡价格并非由两地初始报价唯一确定。

\section{资产定价基本原理}\label{sec:wj-four-pricing}
\subsection{定价算子与一价定律}
% 原书式 (4.1) 用标量 S 表示单只证券价格，此处用 s 与价格向量 S 区分。
\begin{tcolorbox}[knowledge,title=定价算子,breakable=false]
资产定价关系是从支付 $x$ 到价格的映射。记\textbf{定价算子}（估值算子）为 $V$，支付 $x$ 的标量价格为 $s$：
\begin{equation}
    s=V(x).
 \label{eq:wj-four-1}
\end{equation}
\end{tcolorbox}

在\textbf{可达支付空间} $\mathcal M=\{X\theta:\theta\in\mathrm R^N\}$ 上定义 $V(X\theta)=S^{\mathsf T}\theta$。若同一支付的不同复制组合成本不同，这个映射就没有唯一的值；一价定律正是保证它定义良好的条件。第 $n$ 只证券的价格即 $S_n=V(X_n)$。无套利使该算子具有线性和正性。

\begin{tcolorbox}[knowledge]
\WJFourTheorem{thm:wj-four-3}\textbf{（一价定律）}无套利时，相同支付具有相同价格：
\begin{equation}
    x=y\implies V(x)=V(y).
 \label{eq:wj-four-2}
\end{equation}
\end{tcolorbox}

\textbf{证明。}若 $x=y$ 却 $V(x)>V(y)$，卖出 $x$、买入 $y$，未来净支付为零而初始净收入为正，构成第 1 类套利。因此，同支付同价格排除了第 1 类套利，并给出 $V(0)=0$。对非负、非零支付的价格约束由下面的正性条件给出。

\subsection{正性、单调性与线性}
\begin{tcolorbox}[knowledge]
\WJFourTheorem{thm:wj-four-4}无套利时，非负、非零支付的证券价格严格为正：
\begin{equation}
    x>0\implies V(x)>0.
 \label{eq:wj-four-3}
\end{equation}

\end{tcolorbox}

\begin{tcolorbox}[knowledge]
\WJFourTheorem{thm:wj-four-5}无套利时，定价算子 $V$ 递增：
\begin{equation}
    x_1\ge x_2\implies V(x_1)\ge V(x_2).
 \label{eq:wj-four-4}
\end{equation}
\end{tcolorbox}

若 $x>0$ 而 $V(x)\le0$，买入它即为第 2 或第 3 类套利，故必须 $V(x)>0$。若 $x_1\ge x_2$ 却 $V(x_1)<V(x_2)$，买入 $x_1$、卖出 $x_2$，初始净收入为正且未来支付非负，也构成套利。当 $x_1-x_2>0$ 时还有严格结论 $V(x_1)>V(x_2)$。

\begin{tcolorbox}[knowledge]
\WJFourTheorem{thm:wj-four-6}无摩擦且无套利市场中，可达支付上的定价算子是递增的线性算子：对任意 $a,b\in\mathrm{R}$ 以及支付 $x$、$y$，
\begin{equation}
    V(ax+by)=aV(x)+bV(y).
 \label{eq:wj-four-5}
\end{equation}

从而 $V(0)=0$，并且存在向量 $\phi\in\mathrm{R}^{\Omega}$ 使 $V(x)=\phi^{\mathsf T}x$。
\end{tcolorbox}

\textbf{线性的证明：}分别复制 $x$、$y$，再持有 $a$ 倍与 $b$ 倍组合，得到支付 $ax+by$，成本为 $aV(x)+bV(y)$；一价定律要求它等于 $V(ax+by)$。允许负系数用到了卖空不受限的无摩擦假设。

\textbf{向量表示。}在 $\mathcal M$ 中取一组基 $b_1,\ldots,b_r$，令 $B=[b_1,\ldots,b_r]$、$v=(V(b_1),\ldots,V(b_r))^{\mathsf T}$。$B$ 列满秩，故 $B^{\mathsf T}\phi=v$ 有解。任意 $x=Ba\in\mathcal M$ 满足
\[
 V(x)=v^{\mathsf T}a=\phi^{\mathsf T}Ba=\phi^{\mathsf T}x.
\]

这一表示把线性定价写成状态支付的加权和。下一定理进一步给出可选取严格正权重的充要条件。

\subsection{资产定价基本定理与 Stiemke 引理}
\begin{tcolorbox}[knowledge]
\WJFourTheorem{thm:wj-four-7}\textbf{（资产定价基本定理）}市场不存在套利机会，当且仅当存在 $\phi\gg 0$，使得每只可交易证券的价格都是
\begin{equation}
    S=X^{\mathsf T}\phi.
 \label{eq:wj-four-6}
\end{equation}
% 教材式 (4.6) 写作 S=(phi^T X)^T，与本式等价。
\end{tcolorbox}

\begin{tcolorbox}[knowledge]
\WJFourLemma{lem:wj-four-stiemke}\textbf{（Stiemke 引理）}设 $m,n$ 为正整数，$A\in\mathrm R^{m\times n}$，以下两种情况恰有一种成立：
\begin{enumerate}
 \item 存在 $\theta\in\mathrm R^n$，使 $A\theta\ge0$ 且 $A\theta\ne0$；
 \item 存在 $y\in\mathrm R^m$，使 $y\gg0$ 且 $A^{\mathsf T}y=0$。
\end{enumerate}
\end{tcolorbox}
% 教材引理 4.1（第 57—58 页）的扩展矩阵形式；一般引理证明见教材附录 F。
\textbf{引理的证明。}若两种情况同时成立，则 $y^{\mathsf T}A\theta$ 既严格为正又等于零，故两者相互排斥。

若第一种情况不成立，令 $L=\operatorname{Im}(A)$，并取单位单纯形
\[
 K=\{z\in\mathrm R^m:z\ge0,\ \mathbf1^{\mathsf T}z=1\}.
\]

此时 $L\cap K=\varnothing$。$K$ 紧且 $L$ 为闭子空间，二者的最短距离在某对 $z^*\in K$、$\ell^*\in L$ 处取得，且距离严格为正。令 $y=z^*-\ell^*$。由正交投影性质，$y\perp L$，所以 $A^{\mathsf T}y=0$。对每个 $z\in K$，沿 $z^*+t(z-z^*)$ 作小幅调整仍在 $K$ 内；距离最小的一阶条件给出
\[
 y^{\mathsf T}(z-z^*)\ge0,
 \qquad y^{\mathsf T}z\ge y^{\mathsf T}z^*=\|y\|^2>0.
\]

依次取 $z$ 为各单位向量，得到 $y_i\ge\|y\|^2>0$，即 $y\gg0$。第二种情况成立，引理得证。
% 上述为教材附录分离结论的有限维几何展开：分离支付子空间与非负单位单纯形。

\textbf{资产定价基本定理的证明。}将引理~\ref{lem:wj-four-stiemke}（Stiemke 引理）用于包含当期现金流的扩展支付矩阵，分别证明两个方向。

\textbf{（严格正的定价向量 $\Rightarrow$ 无套利）}若 $S=X^{\mathsf T}\phi$、$\phi\gg0$，则任意组合满足
\[
 S^{\mathsf T}\theta=\phi^{\mathsf T}X\theta.
\]

当 $X\theta=0$ 时成本为零；当 $X\theta\ge0$ 且 $X\theta\ne0$ 时，右端严格为正，成本也严格为正。这分别排除第 1 类和第 2、3 类套利。

\textbf{（无套利 $\Rightarrow$ 严格正的定价向量）}令
\[
 A=\begin{bmatrix}-S^{\mathsf T}\\X\end{bmatrix}
 \in\mathrm R^{(1+\Omega)\times N}.
\]

$A\theta$ 正是组合的扩展净现金流。无套利排除引理的第一种情况，故存在 $y\gg0$ 使 $A^{\mathsf T}y=0$。将其分为 $y=(y_0,y_1^{\mathsf T})^{\mathsf T}$，其中 $y_0>0$、$y_1\gg0$，则
\[
 -y_0S+X^{\mathsf T}y_1=0.
\]

取 $\phi=y_1/y_0\gg0$，便有 $S=X^{\mathsf T}\phi$，必要性得证。

无套利要求所有证券共享同一支严格正的定价向量。市场不完全时，这样的向量可以有多支；每一支都为已有证券给出相同的价格。

\begin{tcolorbox}[example]
\begin{question}
\WJFourExample{ex:wj-four-2}市场依次加入例~\ref{ex:wj-four-1} 的证券 $A$、$B$、$C$。求每一步能为全部已有证券定价的严格正向量集合，并判断是否存在套利。
\end{question}
\begin{solution}
先只有证券 $A$，价格为 $1$，三个状态下支付均为 $1$。任一满足 $\phi_1+\phi_2+\phi_3=1$ 的 $\phi\gg 0$ 都能为它定价：
\[
    \Phi_A=\{\phi\gg 0:\phi_1+\phi_2+\phi_3=1\}.
\]

再加入证券 $B$，价格为 $1$，支付为 $[0;\,2;\,2]$。能为它定价的是
\[
    \Phi_B=\{\phi\gg 0:2\phi_2+2\phi_3=1\}.
\]

同时给 $A$、$B$ 定价的状态价格是交集
\[
    \Phi_A\cap\Phi_B=\{\phi\gg 0:\phi_1=\tfrac12,\;\phi_2+\phi_3=\tfrac12\},
\]

交集中包含 $(1/2,1/4,1/4)^{\mathsf T}\gg0$，所以 $A$、$B$ 的价格无套利；令 $\phi_2=t$、$\phi_3=1/2-t$，其中 $0<t<1/2$，还可得到多支严格正解。

再加入证券 $C$，价格为 $2$，支付为 $[2;\,0;\,0]$。条件变成 $\phi_1=1$，
\[
    \Phi_C=\{\phi\gg 0:\phi_1=1\}.
\]

三者的交集是空集。不存在能为所有证券定价的 $\phi\gg 0$，与例~\ref{ex:wj-four-1} 中构造的套利一致。
\end{solution}
\end{tcolorbox}

\subsection{完全市场中的唯一性}
\begin{tcolorbox}[knowledge]
\WJFourTheorem{thm:wj-four-8}无套利的完全市场上，状态价格向量唯一。
\end{tcolorbox}

\textbf{证明。}设 $\theta_\omega$ 复制状态 $\omega$ 的阿罗证券，$X\theta_\omega=\mathbf{1}_\omega$。则 $\phi_\omega=S^{\mathsf T}\theta_\omega$，由市场价格唯一给定；即使存在冗余证券而复制组合不唯一，一价定律仍保证所有复制组合成本相同。若有两支不同的定价向量 $\phi,\psi$，必有状态 $\omega$ 满足 $\phi_\omega\ne\psi_\omega$；但该阿罗证券已可复制，两支向量却给同一复制支付不同价格，与一价定律矛盾。方阵可逆时，直接有 $\phi=(X^{\mathsf T})^{-1}S$。因此，完全市场中每个支付的复制成本唯一确定其价格。

\section{风险中性定价和鞅}\label{sec:wj-four-risk-neutral}
\subsection{无风险利率与贴现}
设可交易证券中有一只无风险债券，1 期支付为 $\iota=[1,\ldots,1]^{\mathsf T}$，价格为 $S_1$。由定理~\ref{thm:wj-four-7}（资产定价基本定理），存在 $\phi\gg 0$ 使
\[
    S_1=\phi^{\mathsf T}\iota=\sum_{\omega=1}^{\Omega}\phi_{\omega}.
\]
\begin{tcolorbox}[knowledge,title=无风险利率与折现因子,breakable=false]
$1$ 单位债券投资的净收益率称为\textbf{无风险利率} $r_F$：
\[
    S_1(1+r_F)=1,\qquad r_F=\frac{1-S_1}{S_1}.
\]

因而
\begin{equation}
    \frac{1}{1+r_F}=\sum_{\omega=1}^{\Omega}\phi_{\omega}.
 \label{eq:wj-four-7}
\end{equation}
\end{tcolorbox}

用一单位当期消费品买入无风险债券，到期可确定获得 $1+r_F$ 单位消费品。反过来，到期一单位确定支付的现值为 $1/(1+r_F)$。这个折现因子也等于各状态价格之和，因为债券在每个状态都支付一单位。

% 来源：教师 PPT 第 27 页。
\begin{tcolorbox}[example,title=课堂例题：从状态价格读出利率]
\begin{question}
设晴、雨两状态价格为 $\phi=(0.3,0.5)^{\mathsf T}$。求单位无风险债券的价格与无风险利率。
\end{question}
\begin{solution}
债券价格为 $S_1=0.3+0.5=0.8$，故 $r_F=1/0.8-1=25\%$。状态价格之和给出折现因子；各状态的相对权重将在下文用于风险支付的估值。
\end{solution}
\end{tcolorbox}

\subsection{风险中性概率与定价公式}\label{sec:wj-four-risk-neutral-probability}
对其余证券 $n=2,\ldots,N$，支付 $x_n=[x_{1n},\ldots,x_{\Omega n}]^{\mathsf T}$，
\[
    S_n=\phi^{\mathsf T}x_n=\sum_{\omega=1}^{\Omega}\phi_{\omega}x_{\omega n}.
\]
\begin{tcolorbox}[knowledge,title=风险中性概率]
将状态价格规范化，定义\textbf{风险中性概率}
\begin{equation}
    q_{\omega}=\frac{\phi_{\omega}}{\sum_{\omega'}\phi_{\omega'}}.
 \label{eq:wj-four-8}
\end{equation}
$q_{\omega}>0$ 且 $\sum q_{\omega}=1$，故 $Q=\{q_{\omega}\}$ 是有限状态空间上的概率测度。
\end{tcolorbox}
\begin{tcolorbox}[knowledge,title=风险中性定价]
定价公式改写成
\begin{equation}
    S_n=S_1\sum_{\omega}q_{\omega}x_{\omega n},
    \qquad
    S_n=\frac{\mathrm{E}^{Q}[\tilde{x}_n]}{1+r_F},\quad n=1,\ldots,N.
 \label{eq:wj-four-9}
\end{equation}
\end{tcolorbox}
式~\eqref{eq:wj-four-9}（风险中性期望贴现定价）就是式~\eqref{eq:wj-four-6}（状态价格加权定价）的另一种写法，称为\textbf{风险中性定价}（risk-neutral pricing），$Q$ 称为\textbf{风险中性测度}。对于价格 $S_n>0$ 的证券，定义净收益率 $\tilde r_n=\tilde x_n/S_n-1$，该式给出 $\mathrm E^Q[\tilde r_n]=r_F$。在 $Q$ 下，证券的期望收益率等于无风险利率。

上例中，规范化状态价格得到 $q_1=0.3/0.8=3/8$、$q_2=0.5/0.8=5/8$。

风险偏好与禀赋影响基础证券的均衡价格。给定这些价格后，无套利通过复制关系确定其他可复制支付的价格，$Q$ 则把价格中的风险调整纳入概率权重。完全市场中 $Q$ 唯一；不完全市场中通常有多支可行 $Q$，它们对可复制支付给出相同价格，对未被复制的支付则可能给出不同估值。

% 来源：教师 PPT 第 30—31 页。
\begin{tcolorbox}[example,title=课堂例题：套餐支付的价格由复制成本决定]
\begin{question}
股票 $A$ 价格为 $10$，两状态支付为 $[20,0]^{\mathsf T}$；债券 $B$ 价格为 $5$，支付为 $[5,5]^{\mathsf T}$。衍生证券 $C$ 支付为 $[25,5]^{\mathsf T}$。求其无套利价格，用风险中性定价验证；若 $C$ 报价为 $20$，应如何套利？
\end{question}
\begin{solution}
$C$ 的支付等于 $A+B$，复制成本为 $10+5=15$，所以无套利价格为 $15$。若报价为 $20$，卖出 $C$ 并买入 $A+B$，初始收入 $5$，未来净支付为零，构成第 1 类套利。

由 $20\phi_1=10$、$5(\phi_1+\phi_2)=5$，得 $\phi_1=\phi_2=1/2$，无风险利率为零，故 $S_C=\mathrm E^Q[\tilde x_C]=(25+5)/2=15$。复制法与风险中性法给出同一价格；这里的 $Q$ 由两只证券的报价确定。
\end{solution}
\end{tcolorbox}

\begin{tcolorbox}[knowledge,title=真实概率与风险中性概率,breakable=false]
真实概率 $P$ 描述状态发生的可能性；$Q$ 由状态价格规范化得到，用于对支付定价。下面的例子展示两种概率权重的区别。
\end{tcolorbox}

\begin{tcolorbox}[example]
\begin{question}
\WJFourExample{ex:wj-four-3}1 期有两个等可能状态 $a$、$b$。
\begin{center}
\begin{tabular}{@{}c@{\quad}c@{}}
\textbf{证券 1} & \textbf{证券 2} \\[0.35em]
\WJEventTree[left={$1$},stem]{1,1} &
\WJEventTree[left={$\tfrac12$},stem]{2,0}
\end{tabular}
\end{center}
比较两只证券在真实概率下的期望支付，求风险中性概率，并验证定价公式。
\end{question}
\begin{solution}
证券 1 是无风险债券，$r_F=0$。在真实概率下两只证券的期望支付都是 $1$，但证券 2 的价格只有证券 1 的一半：期望值相同，确定支付与风险支付的价格不必相同。

由价格得 $\phi_a+\phi_b=1$、$2\phi_a=1/2$，故 $\phi_a=1/4$、$\phi_b=3/4$，$Q=\{1/4,\,3/4\}$。风险中性定价给出
\[
    \frac{\mathrm{E}^{Q}[\tilde{x}_1]}{1+0}
    =\tfrac14\cdot 1+\tfrac34\cdot 1=1,
    \qquad
    \frac{\mathrm{E}^{Q}[\tilde{x}_2]}{1+0}
    =\tfrac14\cdot 2+\tfrac34\cdot 0=\tfrac12,
\]

由此分别得到两只证券的市场价格。
\end{solution}
\end{tcolorbox}

\subsection{债券计价与等价鞅测度}
因为 $S_1=1/(1+r_F)$ 且债券的 1 期支付 $\tilde{x}_1=1$，式~\eqref{eq:wj-four-9}（风险中性期望贴现定价）也可写成
\[
    \frac{S_n}{S_1}=\mathrm{E}^{Q}\Bigl[\frac{\tilde{x}_n}{\tilde{x}_1}\Bigr].
\]
\begin{tcolorbox}[knowledge,title=债券计价与价格的鞅关系,breakable=false]
$S_n$、$S_1$ 以现时消费品计价；$S_n/S_1$ 则是以无风险债券计价的价格。用债券做计价单位时，0 期价格是 $S_n/S_1$，1 期价格是 $\tilde{x}_n$（债券在 1 期的价格恒为 $1$）。时间贴现体现在计价单位中，风险调整则包含在 $Q$ 中。记 $\hat{S}_{n,t}=S_{n,t}/S_{1,t}$（$t=0,1$），则
\begin{equation}
    \hat{S}_{n,0}=\mathrm{E}^{Q}[\hat{S}_{n,1}].
 \label{eq:wj-four-10}
\end{equation}
\end{tcolorbox}

\begin{tcolorbox}[knowledge,breakable=false]
\WJFourDefinition{def:wj-four-3}设 $\mathcal F_t$ 表示截至 $t$ 期已知的信息。随机过程 $z_t$ 若适应该信息、满足 $\mathrm E^Q|z_t|<\infty$，且
\[
 z_t=\mathrm E^Q[z_{t+1}\mid\mathcal F_t],
\]
则称为测度 $Q$ 下的\textbf{鞅}。即在测度 $Q$ 和当前信息下，下一期的预期变动为零；各期实现值仍可随状态变化。
\end{tcolorbox}

\eqref{eq:wj-four-10} 表明，以债券计价的证券价格在 $Q$ 下是鞅。
\begin{tcolorbox}[knowledge,title=等价鞅测度]
若 $Q$ 与真实概率测度 $P$ 有相同的零概率集，则称二者\textbf{等价}。在有限状态空间中，这等价于 $p_\omega=0$ 当且仅当 $q_\omega=0$。本章只列出正概率状态，且 $\phi\gg0$ 给出 $q\gg0$，所以 $Q$ 与 $P$ 等价。使债券计价的证券价格成为鞅的 $Q$ 称为\textbf{等价鞅测度}。
\end{tcolorbox}

% PPT 第 34 页将“低估破产风险后廉价买保险”直接称为套利；改用可复制事件支付给出严格的零成本例子。
\begin{tcolorbox}[example,title=课堂例题：零概率与状态支付]
\begin{question}
设事件 $E$ 满足 $P(E)>0$，且事件支付 $\mathbf1_E$ 可以交易或复制。若以 $Q'(E)=0$ 的概率测度 $Q'$ 代入风险中性定价公式，该支付的价格是多少？是否产生套利？
\end{question}
\begin{solution}
代入定价公式得到
\[
 \frac{\mathrm E^{Q'}[\mathbf1_E]}{1+r_F}=0.
\]

免费买入后，各状态支付非负，事件 $E$ 发生时支付为 $1$，构成第 2 类套利。因此，对这类可交易支付，无套利要求定价权重保留其正概率事件。
\end{solution}
\end{tcolorbox}

本章只有两个时点，式~\eqref{eq:wj-four-10} 描述 $t=0$ 到 $t=1$ 的关系。多期模型中，相应关系采用各期信息下的条件期望；计价单位与中间现金流须一并纳入价格过程。

\section*{本章小结}
\addcontentsline{toc}{section}{本章小结}
\begin{tcolorbox}[summary]
\begin{enumerate}
    \item 无摩擦时，线性无关的证券组合给出市场结构的等价描述。市场完全当且仅当独立支付的证券数等于状态数。
    \item 套利是 $S^{\mathsf T}\theta\le 0$、$X\theta\ge 0$ 且至少一处严格。它由公开的支付和价格确定。
    \item 均衡中不存在套利。前提是市场无摩擦，且至少有人严格偏好非负、非零的消费增加。无套利是均衡的必要条件。
    \item 资产定价基本定理：无套利当且仅当存在 $\phi\gg 0$ 使 $S=X^{\mathsf T}\phi$。完全市场中这个 $\phi$ 唯一。
    \item 风险中性定价 $S_n=\mathrm{E}^{Q}[\tilde{x}_n]/(1+r_F)$ 是同一公式的改写。$Q$ 是规范化的状态价格，真实概率 $P$ 则描述状态发生的可能性。
    \item 用无风险债券计价后，价格在 $Q$ 下是鞅，$Q$ 是等价鞅测度。
\end{enumerate}
\end{tcolorbox}
