% !TeX root = ../../../金融经济学_王江_讲义及习题.tex
% 三轴共用底面角点，绘图高度为 v+3.2。
\begin{tikzpicture}[
 >=Stealth,font=\small,
 x={(3.25cm,-.85cm)},y={(-1.95cm,-1.65cm)},z={(0cm,1.13cm)},
 declare function={
   utilityheight(\m,\s)=3.2-exp(-\m+\s*\s/2);
   pathmean(\t)=(1-\t)*.42+\t*2.6;
   pathrisk(\t)=(1-\t)*1.5+\t*.6;
 },
 mean slice/.style={blue!70!black,very thick},
 risk slice/.style={teal!65!black,very thick},
 chord/.style={red!70!black,thick}]
 \fill[gray!4] (0,0,0)--(2.6,0,0)--(2.6,1.5,0)--(0,1.5,0)--cycle;
 \draw[gray!35] (2.6,0,0)--(2.6,1.5,0)--(0,1.5,0);

 \foreach \i in {0,...,19} {
   \pgfmathsetmacro{\meanLow}{2.6*\i/20}
   \pgfmathsetmacro{\meanHigh}{2.6*(\i+1)/20}
   \foreach \j in {14,13,...,0} {
     \pgfmathsetmacro{\riskLow}{1.5*\j/15}
     \pgfmathsetmacro{\riskHigh}{1.5*(\j+1)/15}
     \pgfmathtruncatemacro{\tone}{5+3*utilityheight((\meanLow+\meanHigh)/2,
                                                 (\riskLow+\riskHigh)/2)}
     \fill[blue!\tone!gray!10]
       (\meanLow,\riskLow,{utilityheight(\meanLow,\riskLow)})--
       (\meanHigh,\riskLow,{utilityheight(\meanHigh,\riskLow)})--
       (\meanHigh,\riskHigh,{utilityheight(\meanHigh,\riskHigh)})--
       (\meanLow,\riskHigh,{utilityheight(\meanLow,\riskHigh)})--cycle;
   }
 }
 \foreach \s in {0,.5,1,1.5}
   \draw[blue!35!gray!70,thin,domain=0:2.6,samples=41]
     plot (\x,\s,{utilityheight(\x,\s)});
 \foreach \m in {0,.65,1.3,1.95,2.6}
   \draw[blue!35!gray!70,thin,domain=0:1.5,samples=41]
     plot (\m,\x,{utilityheight(\m,\x)});

 \draw[->] (0,0,0)--(0,1.75,0)
   node[left,inner sep=3pt,fill=white] {$\sigma_w$};
 \foreach \value in {0.5,1,1.5}
   \draw (0,\value,0)--++(-.045,0,0)
     node[above left,inner sep=1pt,fill=white] {$\value$};
 \draw[->] (0,0,0)--(2.82,0,0)
   node[below right,inner sep=1pt] {平均财富 $\bar w$};
 \foreach \value in {1,2}
   \draw (\value,0,0)--++(0,-.075,0) node[below,inner sep=2pt] {$\value$};
 \draw[->] (0,0,0)--(0,0,3.53) node[above] {$v$};
 \foreach \value in {-3,-2,-1,0}
   \draw (0,0,{\value+3.2})--++(-.045,0,0) node[left,inner sep=2pt] {$\value$};
 \node[below,inner sep=2pt] at (0,0,0) {$0$};

 \draw[mean slice,domain=0:2.6,samples=51]
   plot (\x,{pathrisk(1)},{utilityheight(\x,pathrisk(1))});
 \draw[risk slice,domain=0:1.5,samples=51]
   plot ({pathmean(0)},\x,{utilityheight(pathmean(0),\x)});

 \foreach \point/\t in {P/0,M/.5,Q/1}
   \coordinate (\point) at
     ({pathmean(\t)},{pathrisk(\t)},{utilityheight(pathmean(\t),pathrisk(\t))});
 \coordinate (C) at ($(P)!.5!(Q)$);
 \draw[chord,densely dashed] (P)--(Q);
 \draw[chord,densely dotted] (C)--(M);
 \draw[chord,very thick,domain=0:1,samples=51,variable=\t]
   plot ({pathmean(\t)},{pathrisk(\t)},{utilityheight(pathmean(\t),pathrisk(\t))});
 \fill[red!70!black] (P) circle(1.65pt) node[below left] {$P$};
 \fill[red!70!black] (Q) circle(1.65pt) node[above right] {$Q$};
 \fill[red!70!black] (M) circle(1.65pt) node[above right] {$M$};
 \filldraw[chord,fill=white] (C) circle(1.65pt) node[below right] {$C$};
\end{tikzpicture}
